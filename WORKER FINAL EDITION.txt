export default {
  async fetch(request) {
    try {
      const url = new URL(request.url);

      // رابط ملف M3U الأصلي على GitHub
      const githubM3U = "https://raw.githubusercontent.com/djamekchaouadi/GAMERDZ1517-TV/main/5GAMERDZ1517.m3u";

      const mac = url.searchParams.get("mac") || "00:1A:79:34:62:66";

      // إذا حددنا id → نرسل قناة واحدة (تشغيل مباشر)
      const id = url.searchParams.get("id");
      if (id) {
        // -------------------------------
        // قناة واحدة: توليد رابط مباشر مع Handshake
        // -------------------------------
        let finalUrl = `http://main.light-ott.net:80/play/live.php?mac=${mac}&stream=${id}&extension=ts`;

        try {
          const STALKER = "http://main.light-ott.net:80/portal/server/load.php";
          const hsUrl = `${STALKER}?type=stb&action=handshake&mac=${encodeURIComponent(mac)}&JsHttpRequest=1-xml`;
          const hsResp = await fetch(hsUrl, { headers: { "User-Agent":"Mozilla/5.0","Accept":"application/json" } });
          const hsJson = await hsResp.json().catch(()=>null);
          const token = hsJson?.js?.token;
          if (token) finalUrl += `&token=${token}`;
        } catch {}

        return Response.redirect(finalUrl, 302);
      }

      // -------------------------------
      // قراءة M3U كامل من GitHub
      // -------------------------------
      const res = await fetch(githubM3U);
      const m3uText = await res.text();

      const lines = m3uText.split(/\r?\n/);
      let newM3U = "#EXTM3U\n";

      for (let i = 0; i < lines.length; i++) {
        const line = lines[i].trim();

        // نسخ كل السطور التي تبدأ بـ #EXTINF مباشرة
        if (line.startsWith("#EXTINF")) {
          newM3U += line + "\n";

          // السطر التالي يجب أن يكون الرابط الأصلي
          const urlLine = (lines[i+1] || "").trim();
          const streamMatch = urlLine.match(/stream=(\d+)/);
          if (streamMatch) {
            const streamId = streamMatch[1];
            // رابط جديد → يشير إلى Worker نفسه
            newM3U += `${url.origin}?id=${streamId}&mac=${mac}\n`;
          } else {
            // إذا لم نجد stream → ننسخ الرابط كما هو
            newM3U += urlLine + "\n";
          }

          i++; // تخطّي السطر التالي لأنه تم التعامل معه
        }
      }

      return new Response(newM3U, {
        headers: {
          "Content-Type": "application/x-mpegURL; charset=utf-8",
          "Content-Disposition": "inline; filename=channels.m3u"
        }
      });

    } catch (err) {
      return new Response("Worker error: " + err.message, { status: 500 });
    }
  }
};
